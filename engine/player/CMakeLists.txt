include(FetchContent)

# Third-party notices shipped with exported games: the license text of everything linked into the
# player, collected at configure time from the pinned dependency sources and embedded in the binary.
set(_notices_txt "${CMAKE_CURRENT_BINARY_DIR}/THIRD_PARTY_NOTICES.txt")
file(WRITE ${_notices_txt}
    "Third-party software in this game (VICTUS engine runtime)\n"
    "=========================================================\n\n"
    "This game was made with VICTUS, licensed under the Apache License 2.0.\n"
    "The runtime includes the following third-party components.\n")
function(victus_add_notice title file)
    if(EXISTS "${file}")
        file(READ "${file}" _text)
        file(APPEND ${_notices_txt} "\n------------------------------------------------------------\n${title}\n------------------------------------------------------------\n${_text}\n")
    else()
        file(APPEND ${_notices_txt} "\n------------------------------------------------------------\n${title}\n------------------------------------------------------------\n(license file not available in this build: ${file})\n")
    endif()
endfunction()
foreach(dep sdl3 luau joltphysics box2d miniaudio)
    FetchContent_GetProperties(${dep})
endforeach()
victus_add_notice("VICTUS (Apache-2.0)" "${PROJECT_SOURCE_DIR}/LICENSE")
if(sdl3_SOURCE_DIR)
    victus_add_notice("SDL3 3.4.20 (zlib)" "${sdl3_SOURCE_DIR}/LICENSE.txt")
else()
    file(APPEND ${_notices_txt} "\n------------------------------------------------------------\nSDL3 (zlib)\n------------------------------------------------------------\nSystem-provided SDL3; see its distribution for the license text.\n")
endif()
victus_add_notice("Luau 0.747 (MIT)" "${luau_SOURCE_DIR}/LICENSE.txt")
victus_add_notice("Lua 5.1 (MIT, included in Luau)" "${luau_SOURCE_DIR}/lua_LICENSE.txt")
victus_add_notice("Jolt Physics 5.7.0 (MIT)" "${joltphysics_SOURCE_DIR}/LICENSE")
victus_add_notice("Box2D 3.2.0 (MIT)" "${box2d_SOURCE_DIR}/LICENSE")
victus_add_notice("miniaudio 0.11.26 (public domain or MIT-0)" "${miniaudio_SOURCE_DIR}/LICENSE")
victus_add_notice("stb (stb_image, stb_image_write, stb_truetype; MIT or public domain)" "${PROJECT_SOURCE_DIR}/third_party/stb/LICENSE")
file(APPEND ${_notices_txt} "\nstb_vorbis (bundled with miniaudio, MIT or public domain): Copyright (c) 2017 Sean Barrett.\n")
victus_add_notice("cgltf 1.15 (MIT)" "${PROJECT_SOURCE_DIR}/third_party/cgltf/LICENSE")
victus_add_notice("Dear ImGui 1.93.0 (MIT)" "${PROJECT_SOURCE_DIR}/third_party/imgui/LICENSE.txt")
victus_add_notice("Inter font 5.2.8 (SIL Open Font License 1.1)" "${PROJECT_SOURCE_DIR}/third_party/fonts/inter/LICENSE.txt")

file(READ ${_notices_txt} _hex HEX)
string(LENGTH "${_hex}" _hex_len)
math(EXPR _bytes "${_hex_len} / 2")
string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," _array "${_hex}")
set(_notices_cpp "${CMAKE_CURRENT_BINARY_DIR}/notices.cpp")
file(WRITE ${_notices_cpp}.tmp
    "// Generated by engine/player/CMakeLists.txt from the dependency license files. Do not edit.\n"
    "#include \"victus/player/export.hpp\"\n"
    "namespace {\nconst unsigned char kNotices[] = {${_array}0x00};\n}\n"
    "std::string_view victus::player::third_party_notices() noexcept {\n"
    "    return {reinterpret_cast<const char*>(kNotices), ${_bytes}};\n}\n")
file(COPY_FILE ${_notices_cpp}.tmp ${_notices_cpp} ONLY_IF_DIFFERENT)

victus_add_module(player
    SOURCES
        src/player.cpp
        src/export.cpp
        ${_notices_cpp}
    DEPS VICTUS::Runtime VICTUS::Project VICTUS::Gameplay VICTUS::Script VICTUS::Platform VICTUS::Audio)
