include(FetchContent)

# Third-party notices shipped with exported games: the license text of everything linked into the
# player, collected at configure time from the pinned dependency sources and embedded in the binary.
set(_notices_txt "${CMAKE_CURRENT_BINARY_DIR}/THIRD_PARTY_NOTICES.txt")
file(WRITE ${_notices_txt}
    "Third-party software in this game (VICTUS engine runtime)\n"
    "=========================================================\n\n"
    "This game was made with VICTUS, licensed under the Apache License 2.0.\n"
    "The runtime includes the following third-party components.\n")
function(victus_add_notice title file)
    if(EXISTS "${file}")
        file(READ "${file}" _text)
        file(APPEND ${_notices_txt} "\n------------------------------------------------------------\n${title}\n------------------------------------------------------------\n${_text}\n")
    else()
        file(APPEND ${_notices_txt} "\n------------------------------------------------------------\n${title}\n------------------------------------------------------------\n(license file not available in this build: ${file})\n")
    endif()
endfunction()

file(READ ${_notices_txt} _hex HEX)
string(LENGTH "${_hex}" _hex_len)
math(EXPR _bytes "${_hex_len} / 2")
string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," _array "${_hex}")
set(_notices_cpp "${CMAKE_CURRENT_BINARY_DIR}/notices.cpp")
file(WRITE ${_notices_cpp}.tmp
    "// Generated by engine/player/CMakeLists.txt from the dependency license files. Do not edit.\n"
    "#include \"victus/player/export.hpp\"\n"
    "namespace {\nconst unsigned char kNotices[] = {${_array}0x00};\n}\n"
    "std::string_view victus::player::third_party_notices() noexcept {\n"
    "    return {reinterpret_cast<const char*>(kNotices), ${_bytes}};\n}\n")
file(COPY_FILE ${_notices_cpp}.tmp ${_notices_cpp} ONLY_IF_DIFFERENT)
