# Native plugin host (ADR-0078). Plugin authors use only include/victus/plugin/victus_plugin.h.
# `plugin::scaffold` (victus plugins --new, the Studio Plugins panel) writes the SDK header into
# each new plugin so it builds without an engine checkout. Embedded at configure time.
set(_sdk_header "${CMAKE_CURRENT_SOURCE_DIR}/include/victus/plugin/victus_plugin.h")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${_sdk_header}")
file(READ "${_sdk_header}" _sdk_text)
set(_sdk_cpp "${CMAKE_CURRENT_BINARY_DIR}/plugin_sdk_header.cpp")
file(WRITE "${_sdk_cpp}.tmp"
    "// Generated by engine/plugin/CMakeLists.txt from include/victus/plugin/victus_plugin.h.\n"
    "#include <string_view>\n"
    "namespace victus::plugin { extern const std::string_view kSdkHeader; }\n"
    "const std::string_view victus::plugin::kSdkHeader = R\"VICTUS_SDK(${_sdk_text})VICTUS_SDK\";\n")
file(COPY_FILE "${_sdk_cpp}.tmp" "${_sdk_cpp}" ONLY_IF_DIFFERENT)

victus_add_module(plugin
    SOURCES
        src/plugin.cpp
        src/scaffold.cpp
        "${_sdk_cpp}"
    DEPS VICTUS::Gameplay VICTUS::Scene
    PRIVATE_DEPS ${CMAKE_DL_LIBS})

# Header-only C SDK target for out-of-tree plugin libraries (no engine code is linked).
add_library(victus_plugin_sdk INTERFACE)
add_library(VICTUS::PluginSdk ALIAS victus_plugin_sdk)
target_include_directories(victus_plugin_sdk INTERFACE
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
    $<INSTALL_INTERFACE:include>)
