# Shader sources embedded for project surface shaders compiled at run time: pbr.frag (ADR-0035)
# and the surface vertex template (ADR-0170).
function(victus_embed_text src out symbol)
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${src}")
    file(READ "${src}" _hex HEX)
    string(LENGTH "${_hex}" _len)
    math(EXPR _bytes "${_len} / 2")
    string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," _array "${_hex}")
    file(WRITE ${out}.tmp "// Generated by engine/render/CMakeLists.txt from ${symbol}'s shader. Do not edit.\n"
        "#include <string_view>\nnamespace victus::render::detail {\n"
        "namespace { const char kText[] = {${_array}}; }\n"
        "std::string_view ${symbol}() noexcept { return {kText, ${_bytes}}; }\n}\n")
    file(COPY_FILE ${out}.tmp ${out} ONLY_IF_DIFFERENT)
endfunction()
set(_pbr_cpp "${CMAKE_CURRENT_BINARY_DIR}/pbr_source.cpp")
victus_embed_text("${CMAKE_CURRENT_SOURCE_DIR}/shaders/pbr.frag" ${_pbr_cpp} pbr_source)
set(_vertex_cpp "${CMAKE_CURRENT_BINARY_DIR}/surface_vertex_source.cpp")
victus_embed_text("${CMAKE_CURRENT_SOURCE_DIR}/shaders/templates/surface_vertex.glsl" ${_vertex_cpp} surface_vertex_source_text)

add_library(victus_stb STATIC src/stb_impl.cpp)
target_include_directories(victus_stb PUBLIC ${PROJECT_SOURCE_DIR}/third_party/stb)
# NVIDIA FLIP (BSD-3-Clause, third_party/flip): third-party code, no VICTUS warning flags; always
# optimised because the per-pixel filters are slow in Debug builds.
add_library(victus_flip STATIC ${PROJECT_SOURCE_DIR}/third_party/flip/flip_impl.cpp)
target_compile_features(victus_flip PRIVATE cxx_std_20)
set_target_properties(victus_flip PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(victus_flip SYSTEM PRIVATE ${PROJECT_SOURCE_DIR}/third_party/flip)

    target_compile_options(victus_flip PRIVATE -O2 -w)
endif()
