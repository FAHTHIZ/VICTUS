# pbr.frag's source, embedded for project surface shaders compiled at run time (ADR-0035).
set(_pbr_src "${CMAKE_CURRENT_SOURCE_DIR}/shaders/pbr.frag")
set(_pbr_cpp "${CMAKE_CURRENT_BINARY_DIR}/pbr_source.cpp")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${_pbr_src}")
file(READ "${_pbr_src}" _hex HEX)
string(LENGTH "${_hex}" _len)
math(EXPR _bytes "${_len} / 2")
string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," _array "${_hex}")
file(WRITE ${_pbr_cpp}.tmp "// Generated by engine/render/CMakeLists.txt from shaders/pbr.frag. Do not edit.\n"
    "#include <string_view>\nnamespace victus::render::detail {\n"
    "namespace { const char kPbr[] = {${_array}}; }\n"
    "std::string_view pbr_source() noexcept { return {kPbr, ${_bytes}}; }\n}\n")
file(COPY_FILE ${_pbr_cpp}.tmp ${_pbr_cpp} ONLY_IF_DIFFERENT)

victus_add_module(render
    SOURCES
        src/builtin_shaders_lookup.cpp
        src/environment_map.cpp
        src/atmosphere.cpp
        src/temporal.cpp
        src/light_clusters.cpp
        src/reflection_capture.cpp
        src/irradiance_volume.cpp
        src/irradiance_bake.cpp
        src/gpu_context.cpp
        src/image.cpp
        src/image_compare.cpp
        src/render_path.cpp
        src/mesh.cpp
        src/mesh_optimize.cpp
        src/render_graph.cpp
        src/scene_renderer.cpp
        src/sprite_batch.cpp
        src/surface_shader.cpp
        ${_pbr_cpp}
        generated/builtin_shaders.cpp
    DEPS VICTUS::Core VICTUS::Rhi)
# stb implementations are third-party code: compiled without VICTUS warning flags.
add_library(victus_stb STATIC src/stb_impl.cpp)
target_include_directories(victus_stb PUBLIC ${PROJECT_SOURCE_DIR}/third_party/stb)
# NVIDIA FLIP (BSD-3-Clause, third_party/flip): third-party code, no VICTUS warning flags; always
# optimised because the per-pixel filters are slow in Debug builds.
add_library(victus_flip STATIC ${PROJECT_SOURCE_DIR}/third_party/flip/flip_impl.cpp)
target_compile_features(victus_flip PRIVATE cxx_std_20)
set_target_properties(victus_flip PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(victus_flip SYSTEM PRIVATE ${PROJECT_SOURCE_DIR}/third_party/flip)
if(NOT MSVC)
    target_compile_options(victus_flip PRIVATE -O2 -w)
endif()
target_link_libraries(victus_render PUBLIC victus_stb PRIVATE victus_meshoptimizer victus_flip)
