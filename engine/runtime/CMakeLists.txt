# Default game UI fonts (Inter, SIL OFL 1.1, third_party/fonts/inter) embedded as byte arrays.
set(_fonts_cpp "${CMAKE_CURRENT_BINARY_DIR}/builtin_fonts.cpp")
set(_fonts_src "// Generated by engine/runtime/CMakeLists.txt from third_party/fonts. Do not edit.\n#include <cstddef>\n#include <span>\nnamespace victus::runtime::detail {\n")
# Per weight: the Inter Latin subset plus Latin Extended, Cyrillic and Greek, then Noto Sans
# Bengali and Arabic (SIL OFL 1.1, third_party/fonts/noto) as shaping fallbacks (ADR-0224).
set(_font_files)
foreach(_font Medium Bold Medium-latin-ext Bold-latin-ext Medium-cyrillic Bold-cyrillic Medium-greek Bold-greek)
    list(APPEND _font_files "inter/Inter-${_font}.ttf|inter_${_font}")
endforeach()
foreach(_script Bengali Arabic)
    foreach(_weight Medium Bold)
        list(APPEND _font_files "noto/NotoSans${_script}-${_weight}.ttf|noto_${_script}_${_weight}")
    endforeach()
endforeach()
foreach(_entry IN LISTS _font_files)
    string(REPLACE "|" ";" _parts "${_entry}")
    list(GET _parts 0 _rel)
    list(GET _parts 1 _sym)
    set(_file "${PROJECT_SOURCE_DIR}/third_party/fonts/${_rel}")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${_file}")
    file(READ "${_file}" _hex HEX)
    string(LENGTH "${_hex}" _len)
    math(EXPR _bytes "${_len} / 2")
    string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," _array "${_hex}")
    string(REPLACE "-" "_" _sym "${_sym}")
    string(APPEND _fonts_src "namespace { const unsigned char k_${_sym}[] = {${_array}}; }\n"
        "std::span<const unsigned char> ${_sym}() noexcept { return {k_${_sym}, ${_bytes}}; }\n")
endforeach()
string(APPEND _fonts_src "} // namespace victus::runtime::detail\n")
file(WRITE ${_fonts_cpp}.tmp "${_fonts_src}")
file(COPY_FILE ${_fonts_cpp}.tmp ${_fonts_cpp} ONLY_IF_DIFFERENT)

victus_add_module(runtime
    SOURCES
        src/scene_rendering.cpp
        src/scene_lightmap.cpp
        src/scene_view.cpp
        src/game_ui.cpp
        src/hot_reload.cpp
        src/surface_shader.cpp
        src/world_streaming.cpp
        src/physics_ground_probe.cpp
        ${_fonts_cpp}
    DEPS VICTUS::Assets VICTUS::Scene VICTUS::Gameplay VICTUS::Animation VICTUS::Physics VICTUS::Particles VICTUS::Gameui VICTUS::Ui VICTUS::Shadercompiler VICTUS::Tilemap VICTUS::Terrain VICTUS::Graph VICTUS::World VICTUS::Text)
