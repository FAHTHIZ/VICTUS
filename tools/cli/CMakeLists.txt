add_executable(victus_cli main.cpp)
set_target_properties(victus_cli PROPERTIES OUTPUT_NAME victus)
target_link_libraries(victus_cli PRIVATE VICTUS::Player VICTUS::Assets VICTUS::Project VICTUS::Script VICTUS::Gameplay VICTUS::Net)
victus_configure_target(victus_cli)
# `victus export` ships victus_player from the same folder by default.
add_dependencies(victus_cli victus_player_app)
# `victus plugins --new` writes the plugin SDK header into the new plugin, so a plugin builds
# without an engine checkout (ADR-0078). Embedded at configure time; rebuilt when it changes.
set(_sdk_header "${PROJECT_SOURCE_DIR}/engine/plugin/include/victus/plugin/victus_plugin.h")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${_sdk_header}")
file(READ "${_sdk_header}" _sdk_text)
set(_sdk_cpp "${CMAKE_CURRENT_BINARY_DIR}/plugin_sdk_header.cpp")
file(WRITE "${_sdk_cpp}.tmp"
    "// Generated by tools/cli/CMakeLists.txt from engine/plugin/include/victus/plugin/victus_plugin.h.\n"
    "#include <string_view>\n"
    "namespace victus_cli { extern const std::string_view kPluginSdkHeader; }\n"
    "const std::string_view victus_cli::kPluginSdkHeader = R\"VICTUS_SDK(${_sdk_text})VICTUS_SDK\";\n")
file(COPY_FILE "${_sdk_cpp}.tmp" "${_sdk_cpp}" ONLY_IF_DIFFERENT)
target_sources(victus_cli PRIVATE "${_sdk_cpp}")
