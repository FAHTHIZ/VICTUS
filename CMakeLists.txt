cmake_minimum_required(VERSION 3.24)
project(VICTUS VERSION 0.2.0 DESCRIPTION "Independent cross-platform game engine" LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
# VICTUS uses no C++20 modules. Without this, CMake scans every C++20 source for module
# dependencies with whatever clang-scan-deps it finds; on Android CI it picked the host's
# clang-scan-deps-21, which cannot read the NDK compiler's precompiled headers (SDL3).
set(CMAKE_CXX_SCAN_FOR_MODULES OFF)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
list(APPEND CMAKE_MODULE_PATH ${CMAKE_CURRENT_SOURCE_DIR}/cmake)

option(VICTUS_BUILD_TESTS "Build VICTUS automated tests" ON)
option(VICTUS_BUILD_SAMPLES "Build VICTUS sample applications" ON)
option(VICTUS_BUILD_BENCHMARKS "Build VICTUS benchmark executables" ON)
option(VICTUS_WARNINGS_AS_ERRORS "Treat project warnings as errors" OFF)
option(VICTUS_WITH_GRAPHICS "Build platform, RHI, renderer and graphical tools (requires SDL3)" ON)
# The editor, standalone player and `victus` CLI are host tools; mobile builds ship only the engine libraries
# that a game links into its own shared library (docs/ANDROID.md).
if(ANDROID OR IOS)
    set(_victus_tools_default OFF)
else()
    set(_victus_tools_default ON)
endif()
option(VICTUS_BUILD_TOOLS "Build the editor, the standalone player and the victus CLI" ${_victus_tools_default})
set(VICTUS_SANITIZE "" CACHE STRING "Sanitizers for VICTUS targets: '', 'address,undefined', 'thread'")

include(VictusTargets)
include(VictusDependencies)

if(NOT CMAKE_RUNTIME_OUTPUT_DIRECTORY)
    set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)
endif()

# Core packs (`.vpak`) are compressed with zstd and signed with Monocypher's Ed25519 (ADR-0154).
victus_fetch_monocypher()
victus_fetch_zstd()
add_subdirectory(engine/core)
add_subdirectory(engine/ecs)
add_subdirectory(engine/graph)
add_subdirectory(engine/scene)
add_subdirectory(engine/world)
add_subdirectory(engine/gameplay)
add_subdirectory(engine/plugin)
add_subdirectory(engine/animation)
add_subdirectory(engine/particles)
add_subdirectory(engine/gameui)
victus_fetch_jolt()
victus_fetch_box2d()
add_subdirectory(engine/physics)
add_subdirectory(engine/tilemap)
add_subdirectory(engine/terrain)
victus_fetch_recast()
add_subdirectory(engine/navigation)
add_subdirectory(engine/traffic)
add_subdirectory(engine/ai)
victus_fetch_enet()
add_subdirectory(engine/net)
victus_fetch_miniaudio()
add_subdirectory(engine/audio)
add_subdirectory(engine/sequencer)
victus_fetch_luau()
add_subdirectory(engine/script)
add_subdirectory(engine/project)
if(VICTUS_WITH_GRAPHICS)
    victus_fetch_sdl3()
    add_subdirectory(engine/platform)
    add_subdirectory(engine/rhi)
    victus_fetch_meshoptimizer()
    victus_fetch_xatlas()
    add_subdirectory(engine/render)
    victus_fetch_glslang()
    victus_fetch_spirv_cross()
    add_subdirectory(engine/shadercompiler)
    victus_fetch_bc7enc()
    victus_fetch_astcenc()
    add_subdirectory(engine/assets)
    add_subdirectory(engine/runtime)
    add_subdirectory(engine/ui)
    add_subdirectory(engine/player)
    if(VICTUS_BUILD_TOOLS)
        add_subdirectory(engine/library)
        add_subdirectory(editor)
        add_subdirectory(apps/player)
        add_subdirectory(tools/cli)
    endif()
    if(ANDROID)
        add_subdirectory(apps/android)
    endif()
endif()

if(VICTUS_BUILD_TESTS)
    enable_testing()
    add_subdirectory(tests)
endif()
if(VICTUS_BUILD_BENCHMARKS)
    add_subdirectory(benchmarks)
endif()
if(VICTUS_WITH_GRAPHICS AND VICTUS_BUILD_SAMPLES)
    add_subdirectory(samples)
endif()
